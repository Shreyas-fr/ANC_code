import time
import json
import numpy as np
import torch
import soundfile as sf
import os
import sys

from app.stateful_polar_lstm import StatefulPolarLSTM_Wrapper
from app.monitor import PerformanceMonitor

class ANCStream:
    def __init__(self, config_path):
        with open(config_path, 'r') as f:
            self.config = json.load(f)
            
        self.device = torch.device("cpu")
        self.model = StatefulPolarLSTM_Wrapper()
        
        # Load model using map_location='cpu'
        ckpt = torch.load(self.config['model_path'], map_location='cpu')
        if 'model_state_dict' in ckpt:
            self.model.load_state_dict(ckpt['model_state_dict'])
        else:
            self.model.load_state_dict(ckpt)
            
        self.model.to(self.device)
        self.model.eval()
        
        self.monitor = PerformanceMonitor(
            log_path=os.path.join(self.config['log_directory'], "runtime_metrics.csv"),
            deadline_ms=self.config['frame_ms']
        )
        
        self.reset_state()

    def reset_state(self):
        self.h = None
        self.c = None
        self.in_buffer = np.zeros(512, dtype=np.float32)
        self.out_buffer = np.zeros(512, dtype=np.float32)
        self.window = torch.hann_window(512).numpy()
        self.monitor.record_anomaly(state_reset=True)

    def process_frame(self, audio_chunk):
        # audio_chunk: (256,) float32
        start_time = time.time()
        
        try:
            if np.isnan(audio_chunk).any() or np.isinf(audio_chunk).any():
                self.monitor.record_anomaly(nan_inf=True)
                self.reset_state()
                return audio_chunk # Fallback
                
            # Shift in_buffer and append chunk
            self.in_buffer[:256] = self.in_buffer[256:]
            self.in_buffer[256:] = audio_chunk
            
            # STFT (1 frame)
            frame = self.in_buffer * self.window
            stft = np.fft.rfft(frame)
            features = np.concatenate([stft.real, stft.imag])
            
            input_t = torch.tensor(features, dtype=torch.float32).unsqueeze(0).unsqueeze(0) # (1, 1, 514)
            
            with torch.no_grad():
                if self.h is None:
                    self.h = torch.zeros(2, 1, 256, device=self.device)
                    self.c = torch.zeros(2, 1, 256, device=self.device)
                
                mr, mi, self.h, self.c, _, _ = self.model.core(input_t, self.h, self.c)

                
            mr = mr.squeeze().numpy()
            mi = mi.squeeze().numpy()
            
            if np.isnan(mr).any() or np.isinf(mr).any():
                self.monitor.record_anomaly(nan_inf=True)
                self.reset_state()
                return audio_chunk # Fallback
                
            # Apply mask
            mask = mr + 1j * mi
            enh_stft = stft * mask
            
            # iSTFT (1 frame)
            enh_frame = np.fft.irfft(enh_stft, n=512) * self.window
            
            # Overlap-add
            self.out_buffer[:256] += enh_frame[:256]
            out_chunk = self.out_buffer[:256].copy()
            self.out_buffer[:256] = self.out_buffer[256:] + enh_frame[256:]
            self.out_buffer[256:] = 0.0
            
        except Exception as e:
            print(f"Exception in process_frame: {e}")
            self.monitor.record_anomaly(dropped=True)
            return audio_chunk # Fallback
            
        proc_time_ms = (time.time() - start_time) * 1000
        self.monitor.record_frame(proc_time_ms)
        
        return out_chunk

    def run_file(self, in_file, out_file):
        audio, sr = sf.read(in_file)
        if sr != self.config['sample_rate']:
            raise ValueError(f"Expected {self.config['sample_rate']} Hz, got {sr}")
            
        hop = self.config['hop_length']
        out_audio = []
        
        # Process hop by hop continuously
        for i in range(0, len(audio), hop):
            chunk = audio[i:i+hop]
            if len(chunk) < hop:
                chunk = np.pad(chunk, (0, hop - len(chunk)))
            out = self.process_frame(chunk)
            out_audio.extend(out)
            
            if i % (hop * 100) == 0:
                self.monitor.flush_log()
                
        self.monitor.flush_log(force=True)
        sf.write(out_file, out_audio[:len(audio)], sr)
